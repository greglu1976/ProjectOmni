\color{unidarkgreen}{\subsection{Логика отключения В ВН, СН}}
\color{black}

\begin{enumerate}[label=\arabic{section}.\arabic{subsection}.\arabic*,  align=left, labelsep=4pt, leftmargin=0pt, itemindent=57pt]

\item
Логика отключения В1(ЛВ)ВН аналогична В2(ОВ)ВН, В3 ВН, В1 ЛВ(СН), В2 ОВ(СН).

\item
Логика отключения В1(ЛВ) выводится накладкой \textbf{«ЛО В1(ЛВ)ВН»}. 

\item
Цепь отключения В1(ЛВ) выводится накладкой \textbf{«ЦО В1(ЛВ)ВН»}. Время возврата импульса на
отключение регулируется уставкой \textbf{«Tвозвр откл В1(ЛВ)ВН»}. На отключение В1(ЛВ) формируются при
возникновении одного из следующих событий:
\begin{itemize}
\item ЛО осн. РЗ;
\item ЛО рез. РЗ;
\item ЛО внеш РЗ ВН;
\item ЛО внеш АПТ;
\item ТЗО СН (ТЗО-ф);
\item ТЗО СН (ТЗО-н);
\item ЛО внеш РЗ СН;
\item ЛО внеш РЗ НН.
\end{itemize}

\item
Запрет АПВ В1(ЛВ) выводится накладкой \textbf{«ЗАПВ В1(ЛВ)ВН»}. Длительной импульса времени
запрета ПАВ регулируется уставкой \textbf{«Тимп ЗАПВ В1(ЛВ)ВН»}. Запрет АПВ формируются при
возникновении одного из следующих событий:
\begin{itemize}
\item ЛО осн. РЗ;
\item ЛО рез. РЗ;
\item ЛО внеш РЗ ВН;
\item ЛО внеш АПТ;
\item ТЗО СН (ТЗО-ф);
\item ТЗО СН (ТЗО-н);
\item ЗАПВ внеш РЗ ВН;
\item ЗАПВ внеш РЗ СН.
\end{itemize}

\item
Пуск УРОВ НН формируется сигналом \textbf{«Ср. ПО УРОВ НН»}.
