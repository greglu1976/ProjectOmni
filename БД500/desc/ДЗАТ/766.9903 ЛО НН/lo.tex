\color{unidarkgreen}{\subsection{Логика отключения В НН}}
\color{black}

\begin{enumerate}[label=\arabic{section}.\arabic{subsection}.\arabic*,  align=left, labelsep=4pt, leftmargin=0pt, itemindent=57pt]

\item
Логика отключения В НН1 аналогична В НН2.

\item
Логика отключения В НН1 выводится накладкой \textbf{«ЛО НН1»}. 

\item
Цепь отключения В НН1 выводится накладкой \textbf{«ЦО В НН1»}. Время возврата импульса на
отключение регулируется уставкой \textbf{«Tвозвр откл НН1»}. На отключение В1(ЛВ) формируется при
возникновении одного из следующих событий:
\begin{itemize}
\item ЛО осн. РЗ;
\item ЛО рез. РЗ;
\item ЛО внеш АПТ;
\item ТЗО СН (ТЗО-ф);
\item ТЗО СН (ТЗО-н);
\item ТЗО ВН (ТЗО-ф);
\item ТЗО ВН (ТЗО-н);
\item ДЗО НН;
\item ЛО внеш РЗ ВН;
\item ЛО внеш РЗ СН.
\end{itemize}

\item
Запрет АПВ В НН1 выводится накладкой \textbf{«ЗАПВ В НН1»}. Длительной импульса времени запрета
ПАВ регулируется уставкой \textbf{«Тимп ЗАПВ В НН1»}. Запрет АПВ формируются при
возникновении одного из следующих событий:
\begin{itemize}
\item ЛО осн. РЗ;
\item ЛО рез. РЗ;
\item ЛО внеш АПТ;
\item ТЗО СН (ТЗО-ф);
\item ТЗО СН (ТЗО-н);
\item ТЗО ВН (ТЗО-ф);
\item ТЗО ВН (ТЗО-н);
\item ДЗО НН;
\item Сраб.УРОВ НН1 (Сраб.УРОВ НН2 для В НН2);
\item Сраб.ЗДЗ НН1 (Сраб.ЗДЗ НН2 для В НН2);
\item Ср. МТЗ-2 НН(ОбЩ ЛОГ);
\item ЛО внеш РЗ ВН;
\item ЛО внеш РЗ СН;
\item ЗАПВ внеш РЗ ВН;
\item ЗАПВ внеш РЗ СН.
\end{itemize}

\item
Запрет АВР НН1 формируется при появлении сигнала \textbf{«Ср. МТЗ-2 НН»}.
