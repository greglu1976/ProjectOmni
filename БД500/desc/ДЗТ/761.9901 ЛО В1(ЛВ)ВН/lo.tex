\color{unidarkgreen}{\subsection{Логика отключения В ВН}}
\color{black}

\subsubsection{Общие сведения}

\begin{enumerate}[label=\arabic{section}.\arabic{subsection}.\arabic{subsubsection}.\arabic*,  align=left, labelsep=4pt, leftmargin=0pt, itemindent=57pt]

\item
Логика отключения В1(ЛВ)ВН аналогична В2(ОВ)ВН, В3 ВН.
\item
Логика отключения В1 НН1(НН) аналогична В2 НН1(НН), В1 НН2(СН), В2 НН2(СН).

\item
Логика отключения В1(ЛВ) выводится накладкой \textbf{«ЛО В1(ЛВ)ВН»}. 

\item
Цепь отключения В1(ЛВ) выводится накладкой \textbf{«ЦО В1(ЛВ)ВН»}. Время возврата импульса на
отключение регулируется уставкой \textbf{«Tвозвр откл В1(ЛВ)ВН»}. На отключение В1(ЛВ) формируются при
возникновении одного из следующих событий:
\begin{itemize}
\item ЛО осн. РЗ;
\item ЛО рез. РЗ;
\item ЛО внеш РЗ ВН;
\item ЛО внеш АПТ;
\item ТЗО СН (ТЗО-ф);
\item ТЗО СН (ТЗО-н);
\item ТЗНП-2 РЗН1);
\item ТЗНП-2 РЗН2);
\item ЛО внеш РЗ СН;
\item ЛО внеш РЗ НН.
\end{itemize}

\item
Запрет АПВ В1(ЛВ) выводится накладкой \textbf{«ЗАПВ В1(ЛВ)ВН»}. Длительной импульса времени
запрета ПАВ регулируется уставкой \textbf{«Тимп ЗАПВ В1(ЛВ)ВН»}. Запрет АПВ формируются при
возникновении одного из следующих событий:
\begin{itemize}
\item ЛО осн. РЗ;
\item ЛО рез. РЗ;
\item ЛО внеш РЗ ВН;
\item ЛО внеш АПТ;
\item ТЗО СН (ТЗО-ф);
\item ТЗО СН (ТЗО-н);
\item ТЗНП-2 РЗН1);
\item ТЗНП-2 РЗН2);
\item ЗАПВ внеш РЗ ВН;
\item ЗАПВ внеш РЗ СН.
\end{itemize}

\item
Пуск УРОВ НН формируются при возникновении одного из следующих событий:
\begin{itemize}
\item Ср. ПО УРОВ НН1(НН);
\item Ср. ПО УРОВ НН2(СН).
\end{itemize}

\item
Логика отключения В1 НН2(СН) выводится накладкой \textbf{«ЛО В1 НН2(СН)»}.

\item
Цепь отключения В1 НН2(СН) выводится накладкой \textbf{«ЦО В1 НН2(СН)»}. Время возврата импульса
на отключение регулируется уставкой \textbf{«Tвозвр откл В1 НН2(СН)»}. На отключение В1 НН2(СН)
формируются при возникновении одного из следующих событий:
\begin{itemize}
\item ЛО осн. РЗ;
\item ЛО рез. РЗ;
\item ЛО внеш РЗ ВН;
\item ЛО внеш АПТ;
\item ТЗО СН (ТЗО-ф);
\item ТЗО СН (ТЗО-н);
\item ТЗО ВН (ТЗО-ф);
\item ТЗО ВН (ТЗО-н);
\item Ср. МТЗ-1 НН2(СН);
\item Откл от УРОВ В1 НН2(СН);
\item Откл от ЗДЗ В1 НН2(СН);
\item ЛО внеш РЗ ВН;
\item ЛО внеш РЗ СН.
\end{itemize}

\item
Запрет АПВ В1 НН2(СН) выводится накладкой \textbf{«ЗАПВ В1 НН2(СН)»}. Длительной импульса
времени запрета ПАВ регулируется уставкой \textbf{«Тимп ЗАПВ В1 НН2(СН)»}. Запрет АПВ формируются при
возникновении одного из следующих событий:
\begin{itemize}
\item ЛО осн. РЗ;
\item ЛО рез. РЗ;
\item ЛО внеш АПТ;
\item ТЗО СН (ТЗО-ф);
\item ТЗО СН (ТЗО-н);
\item ТЗО ВН (ТЗО-ф);
\item ТЗО ВН (ТЗО-н);
\item Ср. МТЗ-1 НН2(СН);
\item Откл от УРОВ В1 НН2(СН);
\item Откл от ЗДЗ В1 НН2(СН);
\item ЗАПВ внеш РЗ ВН;
\item ЗАПВ внеш РЗ СН.
\end{itemize}

\item
Запрет АВР В1 НН2(СН) выводится накладкой \textbf{«ЗАВР В1 НН2(СН)»}. Длительной импульса
времени запрета АВР регулируется уставкой \textbf{«Тимп ЗАВР В1 НН2(СН)»}. Запрет АВР формируются при
возникновении одного из следующих событий:
\begin{itemize}
\item Откл от УРОВ В1 НН2(СН);
\item Откл от ЗДЗ В1 НН2(СН);
\item Ср. ТЗНП-1 РЗН2;
\item Ср. МТЗ НН»(СН).
\end{itemize}

\item
Предусматривается возможность приема сигналов \textit{«Откл. В1(ЛВ)ВН»} обеспечения отключения
выключателей стороны ВН в первую очередь при срабатывании резервных защит стороны ВН.
Аналогично для других выключателей.
