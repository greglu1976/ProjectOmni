\color{uniblue}{\subsection{Логика отключения}}
\color{black}

\begin{enumerate}[label=\arabic{section}.\arabic{subsection}.\arabic{enumi}, labelsep=4pt, leftmargin=0pt, itemindent=57pt, itemsep=0pt, parsep=0pt] % если что можно крутить itemsep


\item
Устройство формирует выходное воздействие на  «Откл. осн. Защ.» по схеме «ИЛИ» в следующих 
случаях: 

\begin{itemize}
    \item Срабатывании ДТЗ;  
    \item Срабатывании ДТЗ НП;
    \item Срабатывании отключающей ступени ЗВП. 
\end{itemize}

\item
Устройство формирует выходное воздействие на  «Откл. рез. Защ.» по схеме «ИЛИ» в следующих 
случаях: 

\begin{itemize}
    \item Срабатывании МТЗ; 
    \item Срабатывании ТЗНП; 
    \item Срабатывании ТЗОП;
    \item Срабатывании ЗПВГ.
\end{itemize}

\item
Устройство формирует выходное воздействие на «Запр. АПВ» по схеме «ИЛИ» в следующих 
случаях: 

\begin{itemize}
    \item Срабатывании ДТЗ;
    \item Срабатывании ДТЗ НП; 
    \item Срабатывании отключающей ступени ЗВП;
    \item Срабатывании МТЗ;
    \item Срабатывании ТЗНП;
    \item Срабатывании ТЗОП;
    \item Откл. авар. Изол. ТТ;
    \item Срабатывание отключающей ступени ЗПВГ; 
    \item Срабатывание ЗМН. 
\end{itemize}

\item
Устройство формирует выходное воздействие на «Откл. реж. Защ.» по схеме «ИЛИ» в 
следующих случаях:

\begin{itemize}
    \item Срабатывание отключающей ступени ЗПВГ; 
    \item Срабатывание ЗМН;
    \item Срабатывание отключающей ступени ЗПН. 
\end{itemize}

\item
Устройство формирует выходное воздействие на  «Откл. БСК» по схеме «ИЛИ» в следующих 
случаях: 

\begin{itemize}
    \item Срабатывании ДТЗ;
    \item Срабатывании ДТЗ НП; 
    \item Срабатывании отключающей ступени ЗВП;
    \item Срабатывании МТЗ;
    \item Срабатывании ТЗНП;
    \item Срабатывании ТЗОП;
    \item Откл. авар. Изол. ТТ;
    \item Срабатывание отключающей ступени ЗПВГ; 
    \item Срабатывание ЗМН. 
    \item Срабатывание отключающей ступени ЗПН; 
    \item Внешнее отключение. 
\end{itemize}

\end{enumerate}
