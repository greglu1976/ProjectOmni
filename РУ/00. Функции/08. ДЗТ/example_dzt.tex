
\color{uniblue}\subsection[Выбор уставок ДЗТ]{Выбор уставок ДЗТ} 
\color{black}
\begin{enumerate}
\item
Параметры из раздела <<Общие уставки>> не требуют расчетов, принимаются следующие значения с учетом исходных данных и заданных параметров:\\
\textbf{Сторона 1: 1} = Предусмотрено;\\
\textbf{Сторона 2: 0} = Не предусмотрено;\\
\textbf{Сторона 3: 1} = Предусмотрено;\\
\textbf{Схема соединения трансформаторов тока стороны 1: 0} = «Звезда»;\\
\textbf{Схема соединения трансформаторов тока стороны 3: 0} = «Звезда»;\\
\textbf{Базисная мощность: 16,0} МВА.
\item
Дополнительные параметры приведения сторон:\\
\textbf{Номинальное напряжение стороны 1: 37,0} кВ;\\
\textbf{Номинальный первичный ток трансформатора тока стороны 1: 400,0} А;\\
\textbf{Номинальный вторичный ток трансформатора тока стороны 1: 5,0} А;\\
\textbf{Компенсация токов 3I0 для стороны 1: 0} = Без компенсации;\\
\textbf{Векторная группа стороны 1: 0};\\
\textbf{Номинальное напряжение стороны 3: 6,3} кВ;\\
\textbf{Номинальный первичный ток трансформатора тока стороны 3: 1500,0} А;\\
\textbf{Номинальный вторичный ток трансформатора тока стороны 3: 5,0} А;\\
\textbf{Компенсация токов 3I0 для стороны 3: 0} = Без компенсации;\\
\textbf{Векторная группа стороны 3: 5}.
\item 
Дифференциальный токовый орган с торможением\\
Параметр \textbf{$I_{\textsl{т1}}$} <<Ток торможения, соответствующий началу первого наклонного участка>> принимается \textbf{равным 1,0}.
Определение тока небаланса по выражению (\ref{eq:dzt1}) для выбора параметра <<Дифференциальный ток срабатывания начального участка>>:
\begin{equation*}
    I_{\textsl{нб.расч.}} =  I_{\textsl{нб.расч.*}}^{'} + I_{\textsl{нб.расч.*}}^{''} + I_{\textsl{нб.расч.*}}^{'''} = 0,1 + 0,12 + 0,03 = 0,25\; (\textsl{о.е.}),
\end{equation*}
где:
\begin{itemize}
    \item составляющая, обусловленная погрешностью ТТ (\ref{eq:dzt2}):
{\begin{equation*}
    I_{\textsl{нб.расч.*}}^{'}=  k_{\textsl{пер}} \cdot k_{\textsl{одн}} \cdot \epsilon \cdot I_{\textsl{расч}} = 1,0 \cdot 1,0 \cdot 0,1 \cdot 1,0 = 0,1\; (\textsl{о.е.}),
\end{equation*}}
    \item составляющая небаланса, вносимая устройством регулирования напряжения трансформатора (\ref{eq:dzt3}):
\begin{equation*}
    I_{\textsl{нб.расч.*}}^{''}= \Delta U_{\textsl{рег}} \cdot I_{\textsl{расч}} = 0,12 \cdot 1,0 = 0,12\; (\textsl{о.е.}),
\end{equation*}
\begin{equation*}
\text{где\quad}  
\Delta U_{\textsl{рег}} = \frac {|8 \cdot 1,5\%|+|8 \cdot (-1,5\%)|}{2 \cdot 100 \%} =0,12\; (\textsl{о.е.}),
\end{equation*}
\item составляющая, обусловленная погрешностью выравнивания токов плеч (\ref{eq:dzt4}):
\begin{equation*}
    I_{\textsl{нб.расч.*}}^{'''}= f_{\textsl{выр}} \cdot I_{\textsl{расч}} = 0,03 \cdot 1,0 = 0,03\; (\textsl{о.е.}),
\end{equation*}
\end{itemize}
По выражению (\ref{eq:dzt5}) определяем параметр <<Дифференциальный ток срабатывания начального участка>>:
\begin{equation*}
    I_{\textsl{ср}}= k_{\textsl{отс}} \cdot I_{\textsl{нб.расч}}=1,2 \cdot 0,25 = 0,3\; (\textsl{о.е.}) ,
\end{equation*}
Параметр \textbf{$I_{\textsl{ср(ДТЗт)}}$} <<Дифференциальный ток срабатывания начального участка>> принимается \textbf{равным 0,3}.


Параметр \textbf{$I_{\textsl{т2}}$} <<Ток торможения, соответствующий началу второго наклонного участка>> принимается \textbf{равным 3,0}.
Определение тока небаланса по выражению (\ref{eq:dzt1}) для выбора параметра <<Коэффициент торможения первого наклонного участка>>:
\begin{equation*}
    I_{\textsl{нб.расч.}} =  I_{\textsl{нб.расч.*}}^{'} + I_{\textsl{нб.расч.*}}^{''} + I_{\textsl{нб.расч.*}}^{'''} = 0,45 + 0,36 + 0,09 = 0,9\; (\textsl{о.е.}),
\end{equation*}
где:
\begin{itemize}
    \item составляющая, обусловленная погрешностью ТТ (\ref{eq:dzt2}):
{\begin{equation*}
    I_{\textsl{нб.расч.*}}^{'}=  k_{\textsl{пер}} \cdot k_{\textsl{одн}} \cdot \epsilon \cdot I_{\textsl{расч}} = 1,5 \cdot 1,0 \cdot 0,1 \cdot 3,0 = 0,45\; (\textsl{о.е.}),
\end{equation*}}
    \item составляющая небаланса, вносимая устройством регулирования напряжения трансформатора (\ref{eq:dzt3}):
\begin{equation*}
    I_{\textsl{нб.расч.*}}^{''}= \Delta U_{\textsl{рег}} \cdot I_{\textsl{расч}} = 0,12 \cdot 3,0 = 0,36\; (\textsl{о.е.}),
\end{equation*}
\item составляющая, обусловленная погрешностью выравнивания токов плеч (\ref{eq:dzt4}):
\begin{equation*}
    I_{\textsl{нб.расч.*}}^{'''}= f_{\textsl{выр}} \cdot I_{\textsl{расч}} = 0,03 \cdot 3,0 = 0,09\; (\textsl{о.е.}),
\end{equation*}
\end{itemize}
По выражению (\ref{eq:dzt5}) определяем параметр <<Дифференциальный ток срабатывания начала второго участка>>:
\begin{equation*}
    I_{\textsl{ср(т2)}}= k_{\textsl{отс}} \cdot I_{\textsl{нб.расч}}=1,3 \cdot 0,9 = 1,17\; (\textsl{о.е.}) ,
\end{equation*}
По выражению (\ref{eq:dzt6}) определяем параметр <<Коэффициент торможения первого наклонного участка>>:
\begin{equation*}
    k_{\textsl{т1}} = \frac {I_{\textsl{ср(т2)}} - I_{\textsl{ср}}}{I_{\textsl{т2}}-I_{\textsl{т1}}} = \frac {1,17 - 0,3}{3,0-1,0}=0,435\; (\textsl{о.е.}),
\end{equation*}
Параметр \textbf{$k_{\textsl{т1}}$} <<Коэффициент торможения первого наклонного участка>> принимается \textbf{равным 0,45}.

Определение тока небаланса по выражению (\ref{eq:dzt1}) для выбора параметра <<Коэффициент торможения второго наклонного участка>>:
\begin{equation*}
    I_{\textsl{нб.расч.}} =  I_{\textsl{нб.расч.*}}^{'} + I_{\textsl{нб.расч.*}}^{''} + I_{\textsl{нб.расч.*}}^{'''} = 6,0 + 2,4 + 0,6 = 9,0\; (\textsl{о.е.}),
\end{equation*}
где:
\begin{itemize}
    \item составляющая, обусловленная погрешностью ТТ (\ref{eq:dzt2}):
{\begin{equation*}
    I_{\textsl{нб.расч.*}}^{'}=  k_{\textsl{пер}} \cdot k_{\textsl{одн}} \cdot \epsilon \cdot I_{\textsl{расч}} = 3,0 \cdot 1,0 \cdot 0,1 \cdot 20,0 = 6,0\; (\textsl{о.е.}),
\end{equation*}}
    \item составляющая небаланса, вносимая устройством регулирования напряжения трансформатора (\ref{eq:dzt3}):
\begin{equation*}
    I_{\textsl{нб.расч.*}}^{''}= \Delta U_{\textsl{рег}} \cdot I_{\textsl{расч}} = 0,12 \cdot 20,0 = 2,4\; (\textsl{о.е.}),
\end{equation*}
\item составляющая, обусловленная погрешностью выравнивания токов плеч (\ref{eq:dzt4}):
\begin{equation*}
    I_{\textsl{нб.расч.*}}^{'''}= f_{\textsl{выр}} \cdot I_{\textsl{расч}} = 0,03 \cdot 20,0 = 0,6\; (\textsl{о.е.}),
\end{equation*}
\end{itemize}
По выражению (\ref{eq:dzt5}) определяем значение расчетного дифференциального тока, соответствующего максимальной предельной кратности ТТ:
\begin{equation*}
    I_{\textsl{ср.макс}}= k_{\textsl{отс}} \cdot I_{\textsl{нб.расч}}=1,3 \cdot 9,0 = 11,7\; (\textsl{о.е.}) ,
\end{equation*}
По выражению (\ref{eq:dzt7}) определяем параметр <<Коэффициент торможения второго наклонного участка>>:
\begin{equation*}
    k_{\textsl{т2}} = \frac {I_{\textsl{ср.макс}} - I_{\textsl{ср(т2)}}}{I_{\textsl{т2макс}}-I_{\textsl{т2}}} = \frac {11,7 - 1,17}{20,0-3,0}=0,619\; (\textsl{о.е.}),
\end{equation*}
Параметр \textbf{$k_{\textsl{т2}}$} <<Коэффициент торможения второго наклонного участка>> принимается \textbf{равным 0,7}.

Коэффициент чувствительности определяется по выражению (\ref{eq:dzt9})
\begin{equation*}
    k_{\textsl{ч}} = \frac {I_{\textsl{дифф}}}{I_{\textsl{ср}}}= \frac {3,67}{0,68}=5,39,
\end{equation*}
Условия по чувствительности выполняются, \textbf{$k_{\textsl{ч}} > 2,0$}. 

\item 
Дифференциальная токовая отсечка

Определение тока небаланса по выражению (\ref{eq:dzt1}) для выбора параметра <<Дифференциальный ток срабатывания отсечки>>:
\begin{equation*}
    I_{\textsl{нб.расч.}} =  I_{\textsl{нб.расч.*}}^{'} + I_{\textsl{нб.расч.*}}^{''} + I_{\textsl{нб.расч.*}}^{'''} = 1,67 + 0,67 + 0,167 = 2,507\; (\textsl{о.е.}),
\end{equation*}
где:
\begin{itemize}
    \item составляющая, обусловленная погрешностью ТТ (\ref{eq:dzt2}):
{\begin{equation*}
    I_{\textsl{нб.расч.*}}^{'}=  k_{\textsl{пер}} \cdot k_{\textsl{одн}} \cdot \epsilon \cdot I_{\textsl{расч}} = 3,0 \cdot 1,0 \cdot 0,1 \cdot 5,58 = 1,67\; (\textsl{о.е.}),
\end{equation*}}
    \item составляющая небаланса, вносимая устройством регулирования напряжения трансформатора (\ref{eq:dzt3}):
\begin{equation*}
    I_{\textsl{нб.расч.*}}^{''}= \Delta U_{\textsl{рег}} \cdot I_{\textsl{расч}} = 0,12 \cdot 5,58 = 0,67\; (\textsl{о.е.}),
\end{equation*}
\item составляющая, обусловленная погрешностью выравнивания токов плеч (\ref{eq:dzt4}):
\begin{equation*}
    I_{\textsl{нб.расч.*}}^{'''}= f_{\textsl{выр}} \cdot I_{\textsl{расч}} = 0,03 \cdot 5,58 = 0,167\; (\textsl{о.е.}),
\end{equation*}
\end{itemize}
По выражению (\ref{eq:dzt5}) определяем параметр <<Дифференциальный ток срабатывания отсечки>>:
\begin{equation*}
    I_{\textsl{ср(ДТО)}}= k_{\textsl{отс}} \cdot I_{\textsl{нб.расч}}=1,5 \cdot 2,507 = 3,76\; (\textsl{о.е.}) ,
\end{equation*}
Поскольку значение параметра <<Дифференциальный ток срабатывания отсечки>> по условиям отстройки от броска тока намагничивания по выражению (\ref{eq:dzt16}) ($I_{\textsl{ср(ДТО)}} \geq 6,0$), выше чем по условию отстройки от тока небаланса при внешнем КЗ в максимальном режиме (($I_{\textsl{ср(ДТО)}}=3,76$), то принимаем параметр \textbf{$I_{\textsl{ср(ДТО)}}$} <<Дифференциальный ток срабатывания отсечки>> \textbf{равным 6,0}.

\end{enumerate}
