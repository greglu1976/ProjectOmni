\item Параметры питающей системы (сторона 37 кВ)
\begin{itemize}
\item Сопротивление системы в максимальном режиме: $X_{\textsl{с,max}}=6,79$ Ом
\item Сопротивление системы в минимальном режиме: $X_{\textsl{с,max}}=16,18$ Ом. 
\end{itemize}

\item
На трансформаторе установлены трансформаторы тока на стороне ВН с коэффициентом трансформации 400/5, на стороне НН с коэффициентом трансформации 1500/5. Трансформаторы тока по сторонам ВН и НН собраны в <<звезду>>, полярными выводами к трансформатору.

\needspace{3\baselineskip}
\color{uniblue}\subsection[Расчет сопротивлений силового трансформатора с учетом РПН]{Расчет сопротивлений силового трансформатора с учетом РПН} 
\color{black}


\begin{enumerate}
\item Базовое реактивное сопротивление трансформатора, приведенное к стороне ВН (37 кВ), на средней отпайке:
\begin{equation*}
X_{\textsl{Т,ср}} = \frac {U_k}{100} \cdot \frac {U^2_{\textsl{Т,ср}}}{S_{\textsl{ном}}} = \frac {10}{100} \cdot {37^2}{16} = 8,556~(\textsl{Ом}),
\end{equation*}

\item Расчет сопротивления с учетом РПН:
\begin{itemize}
    \item минимальное сопротивление (максимальный режим, соответствующий нижнему положение РПН): 
    $X_{\textsl{Т,min}} = X_{\textsl{Т,ср}} \cdot (1-\Delta U_{\textsl{рег}})^2 = 8,556 \cdot (1-0,12)^2=6,626~(\textsl{Ом})$;
    \item максимальное сопротивление (минимальный режим, соответствующий верхнему положение РПН):
    $X_{\textsl{Т,max}} = X_{\textsl{Т,ср}} \cdot (1+\Delta U_{\textsl{рег}})^2 = 8,556 \cdot (1+0,12)^2=10,733~(\textsl{Ом})$.
\end{itemize}

\end{enumerate}


\color{uniblue}\subsection[Расчет токов короткого замыкания]{Расчет токов короткого замыкания} 
\color{black}

\begin{enumerate}

\item В точке К1 (на выводах 37 кВ):
\begin{itemize}
    \item максимальный режим (трехфазное металлическое КЗ): 
    \begin{equation*}
    I^{(3)}_{\textsl{K1,max}} = \frac {U_{\textsl{ном, ВН}}}{\sqrt{3} \cdot X_{\textsl{с,max}}} = \frac {37000}{2 \cdot 6,79} = 3146,3~\textsl{А};
    \end{equation*}
    \item минимальный режим (двухфазное металлическое КЗ):
    \begin{equation*}
    I^{(2)}_{\textsl{K1,min}} = \frac {U_{\textsl{ном, ВН}}}{2 \cdot X_{\textsl{с,min}}} = \frac {37000}{2 \cdot 16,18} = 1143,4~\textsl{А};
    \end{equation*}
\end{itemize}

\item В точке К2 (на выводах 6,3 кВ) -- максимальный режим (трехфазное металлическое КЗ), ток рассчитывается при минимальном сопротивлении системы и минимальном сопротивлении трансформатора для отстройки ДЗТ от сквозных токов на шинах НН.

\begin{enumerate}
    \item Ток, приведенный к стороне ВН (37 кВ):
    \begin{equation*}
    I^{(3)}_{\textsl{K2,max,ВН}} = \frac {U_{\textsl{max, ВН}}}{\sqrt{3} \cdot (X_{\textsl{с,max}}+X_{\textsl{Т,min}}) } = \frac {37000}{\sqrt{3} \cdot (6,79+6,626)} = 1592,3~\textsl{А}.
    \end{equation*}

    \item Фактический ток на стороне НН (6,3 кВ):
    \begin{equation*}
    I^{(3)}_{\textsl{K2,max,НН}}  = I^{(3)}_{\textsl{K2,max,ВН}} \cdot \frac {U_{\textsl{ном, ВН}}}{U_{\textsl{ном, НН}}} = 1592,3 \cdot \frac {37}{6,3} = 9351,6~\textsl{А}.
    \end{equation*}
\end{enumerate}


\item В точке К2 (на выводах 6,3 кВ) -- минимальный режим (двухфазное металлическое КЗ),  ток рассчитывается при максимальном сопротивлении системы и максимальном сопротивлении трансформатора для проверки чувствительности ДТЗ при внутреннем повреждении.

\begin{enumerate}
    \item Минимальный ток двухфазного КЗ, приведенный к стороне ВН (37 кВ):
    \begin{equation*}
    I^{(2)}_{\textsl{K2,min,ВН}} = I_{\textsl{КЗ, min}} \cdot \frac {\sqrt{3}}{2} = \frac {41440}{2\cdot (16,18+10,733)} = 769,9~\textsl{А}.
    \end{equation*}

    \item Фактический ток двухфазного КЗ на стороне НН (6,3 кВ):
    \begin{equation*}
    I^{(2)}_{\textsl{K2,min,НН}}  = 769,9 \cdot \frac {37}{6,3} = 4521,6~\textsl{А}.
    \end{equation*}
\end{enumerate}

\color{uniblue}\subsection[Расчет номинальных токов сторон трансформатора]{Расчет номинальных токов сторон трансформатора} 
\color{black}

\begin{enumerate}
\item Номинальный ток трансформатора со стороны ВН:
\begin{equation*}
    I_{\textsl{баз,ВН}} = \frac{S_{\textsl{баз}}}{\sqrt{3} \cdot U_{\textsl{ном,ВН}}}  = \frac{16000}{\sqrt{3} \cdot 37} = 249,68\; (\textsl{А}),
\end{equation*}
\item Номинальный ток трансформатора со стороны НН:
\begin{equation*}
    I_{\textsl{баз,НН}} = \frac{S_{\textsl{баз}}}{\sqrt{3} \cdot U_{\textsl{ном,НН}}}  = \frac{16000}{\sqrt{3} \cdot 6,3} = 1466,28\; (\textsl{А}),
\end{equation*}

\end{enumerate}
